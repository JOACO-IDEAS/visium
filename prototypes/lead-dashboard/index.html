  // antes de este script para entornos sin bundler, y cambiar este import.
  const { createClient } = await import(
    'https://esm.sh/@supabase/supabase-js@2'
  );

  const supabase = createClient(SUPABASE_URL, SUPABASE_ANON_KEY);

  // ── PROPERTY_NAMES: fallback de nombre comercial por property_id ────────
  // Reemplazar con tu tabla de propiedades cuando esté disponible.
  const PROPERTY_NAMES = {
    'prop-luxury-01':  'Penthouse Torre Libertad',
    'prop-economy-02': 'Dpto 2B Palermo Soho',
  };

  // ── Color por temperatura ───────────────────────────────────────────────
  const TEMP_COLORS = {
    HOT:  '#ef4444',
    WARM: '#f97316',
    MILD: '#eab308',
    COLD: '#3b82f6',
  };

  // ── Alert Engine ─────────────────────────────────────────────────────────
  // HIPÓTESIS A VALIDAR: leads con score > 70 tienen probabilidad de conversión
  // suficiente para justificar una intervención proactiva del agente.
  const ALERT_SCORE_THRESHOLD = 70;
  const pendingAlerts = [];

  function buildAlertPayload(lead) {
    const topSignals = (lead.breakdown?.signals ?? [])
      .filter(s => s.type !== 'DWELL')
      .slice(0, 3)
      .map(s => s.label);

    return {
      type:      'VISIUM_HOT_LEAD',
      timestamp:  new Date().toISOString(),
      algorithm_version: lead.algorithm_version ?? 'v1',
      lead: {
        visitor_id:   lead.visitor_id,
        property_id:  lead.property_id,
        property_name: PROPERTY_NAMES[lead.property_id] ?? lead.property_id,
        score:         lead.score,
        temperature:   lead.temperature,
        score_reliable: lead.score_reliable,
        is_return:     lead.is_return,
        last_seen:     lead.computed_at,
        top_signals:   topSignals,
      },
      // Stub de notificación — reemplazar con llamada real a servicio de mensajería
      notification_stub: {
        channel:  'whatsapp',
        to:       'AGENTE_PHONE_NUMBER',
        message: `🔥 Lead VISIUM — ${PROPERTY_NAMES[lead.property_id] ?? lead.property_id}: Score ${lead.score}/100. ${topSignals.join(', ')}. Contactá ahora.`,
      },
    };
  }

  function evaluateAlerts(leads) {
    const hotLeads = leads.filter(l => l.score > ALERT_SCORE_THRESHOLD && l.score_reliable !== false);
    if (hotLeads.length === 0) return;

    const latest = hotLeads[0];
    const payload = buildAlertPayload(latest);
    pendingAlerts.push(payload);

    // Mostrar banner de alerta
    const payloadStr = JSON.stringify(payload, null, 2);
    document.getElementById('alert-title').textContent =
      `🔥 Lead HOT — Score ${latest.score} · ${PROPERTY_NAMES[latest.property_id] ?? latest.property_id}`;
    document.getElementById('alert-payload').textContent = payloadStr;
    document.getElementById('alert-panel').classList.add('visible');

    // Actualizar modal
    document.getElementById('modal-payload').textContent = payloadStr;
  }

  // ── Renderizado de fila ──────────────────────────────────────────────────
  function renderScoreRing(score, temperature) {
    const color   = TEMP_COLORS[temperature] ?? '#5b70f5';
    const radius  = 16;
    const circ    = 2 * Math.PI * radius;
    const dash    = (score / 100) * circ;
    return `
      

        
        
${Math.round(score)}

      
`;
  }

  function renderSignals(breakdown) {
    const signals = (breakdown?.signals ?? [])
      .filter(s => s.type !== 'DWELL')
      .slice(0, 3);

    if (signals.length === 0) {
      return 'Sin señales activas';
    }

    return `

      ${signals.map(s => `${s.label}`).join('')}
    
`;
  }

  function renderTemperatureBadge(temperature) {
    const icons = { HOT: '🔥', WARM: '🌡', MILD: '💛', COLD: '🔵' };
    return `
      ${icons[temperature] ?? ''} ${temperature}
    `;
  }

  function formatRelativeTime(isoString) {
    if (!isoString) return '—';
    const diff = Date.now() - new Date(isoString).getTime();
    const mins  = Math.floor(diff / 60000);
    const hours = Math.floor(diff / 3600000);
    const days  = Math.floor(diff / 86400000);
    if (mins  < 1)   return 'Justo ahora';
    if (mins  < 60)  return `Hace ${mins}m`;
    if (hours < 24)  return `Hace ${hours}h`;
    return `Hace ${days}d`;
  }

  function renderRow(lead, index) {
    const propertyName = PROPERTY_NAMES[lead.property_id] ?? lead.property_id;
    const visitorShort = lead.visitor_id
      ? lead.visitor_id.split('-').pop()
      : 'anónimo';

    const unreliableTag = lead.score_reliable === false
      ? '⚠ escala'
      : '';

    const returnTag = lead.is_return
      ? '↩ RETORNO'
      : '';

    return `
      
        ${index + 1}
        
          
…${visitorShort}

          ${returnTag}
        
        ${propertyName}
        
          

            ${renderScoreRing(lead.score, lead.temperature)}
            ${unreliableTag}
          

        
        ${renderTemperatureBadge(lead.temperature)}
        ${renderSignals(lead.breakdown)}
        ${formatRelativeTime(lead.computed_at)}
      `;
  }

  // ── Carga de datos ──────────────────────────────────────────────────────
  window.loadLeads = async function () {
    const tbody = document.getElementById('leads-body');
    tbody.innerHTML = ` Actualizando...`;

    const { data: leads, error } = await supabase
      .from('visium_scores')
      .select(`
        session_id, visitor_id, property_id, score, temperature,
        zone_score, conversion_score, score_reliable, is_return,
        algorithm_version, computed_at, breakdown
      `)
      .order('score',       { ascending: false })
      .order('computed_at', { ascending: false })
      .limit(50);

    if (error) {
      tbody.innerHTML = `
        Error al cargar: ${error.message}
      `;
      return;
    }

    if (!leads || leads.length === 0) {
      tbody.innerHTML = `
        Sin sesiones registradas aún. Esperando telemetría del viewer...
      `;
      return;
    }

    // Renderizar filas
    tbody.innerHTML = leads.map((lead, i) => renderRow(lead, i)).join('');

    // KPIs
    document.getElementById('kpi-total').textContent = leads.length;
    document.getElementById('kpi-hot').textContent   = leads.filter(l => l.temperature === 'HOT').length;
    document.getElementById('kpi-warm').textContent  = leads.filter(l => l.temperature === 'WARM').length;
    const avg = leads.reduce((s, l) => s + l.score, 0) / leads.length;
    document.getElementById('kpi-avg').textContent   = avg.toFixed(1);

    // Header
    const now = new Date().toLocaleTimeString('es-AR', { hour: '2-digit', minute: '2-digit' });
    document.getElementById('last-refresh').textContent = `Actualizado: ${now}`;

    // Alert Engine
    evaluateAlerts(leads);
  };

  // ── Alert UI helpers ────────────────────────────────────────────────────
  window.dismissAlert = () => document.getElementById('alert-panel').classList.remove('visible');

  window.copyAlert = () => {
    const text = document.getElementById('alert-payload').textContent;
    navigator.clipboard.writeText(text).then(() => alert('JSON copiado'));
  };

  window.toggleModal = () => {
    document.getElementById('modal-overlay').classList.toggle('open');
  };

  window.closeModalOnOverlay = (e) => {
    if (e.target === document.getElementById('modal-overlay')) toggleModal();
  };

  window.copyModalPayload = () => {
    const text = document.getElementById('modal-payload').textContent;
    navigator.clipboard.writeText(text).then(() => alert('JSON copiado'));
  };

  window.showLeadModal = (sessionId) => {
    // Placeholder: en PR-5 abrirá el detalle de sesión con timeline
    console.log('[VISIUM] Detalle de sesión:', sessionId);
  };

  // ── Bootstrap + auto-refresh ────────────────────────────────────────────
  loadLeads();
  setInterval(loadLeads, 60_000);

  // Subscripción realtime para alertas inmediatas en inserts nuevos
  supabase
    .channel('visium-scores-live')
    .on('postgres_changes',
      { event: 'INSERT', schema: 'public', table: 'visium_scores' },
      (payload) => {
        const newLead = payload.new;
        // Re-cargar tabla y evaluar alerta del nuevo lead
        loadLeads();
        if (newLead.score > ALERT_SCORE_THRESHOLD && newLead.score_reliable !== false) {
          const alertPayload = buildAlertPayload(newLead);
          const payloadStr = JSON.stringify(alertPayload, null, 2);
          document.getElementById('alert-title').textContent =
            `🔥 Nuevo Lead HOT en tiempo real — Score ${newLead.score}`;
          document.getElementById('alert-payload').textContent = payloadStr;
          document.getElementById('alert-panel').classList.add('visible');
          document.getElementById('modal-payload').textContent = payloadStr;
        }
      }
    )
    .subscribe();
